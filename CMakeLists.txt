cmake_minimum_required(VERSION 3.20)
project(Mimic VERSION 1.0.0 LANGUAGES C CXX)
# Mimic: one person's face onto a picture or a video, in C++. Part of Atelier.
#   Mimic      the window (SDL2, OpenGL 3, Dear ImGui)
#   mimic-cli  the command line
# ONNX Runtime (the DirectML build: any DirectX 12 GPU) is loaded at run time; video goes through ffmpeg.
#   Windows: build.bat    Linux: ./build.sh
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(STB_DIR ${CMAKE_SOURCE_DIR}/external/stb)
foreach(f stb_image.h stb_image_write.h)
    if(NOT EXISTS ${STB_DIR}/${f})
        file(DOWNLOAD https://raw.githubusercontent.com/nothings/stb/master/${f} ${STB_DIR}/${f})
    endif()
endforeach()
set(ORT_DIR ${CMAKE_SOURCE_DIR}/external/ortdml)
if(NOT EXISTS ${ORT_DIR}/onnxruntime_c_api.h)
    file(DOWNLOAD https://raw.githubusercontent.com/microsoft/onnxruntime/v1.20.1/include/onnxruntime/core/session/onnxruntime_c_api.h ${ORT_DIR}/onnxruntime_c_api.h)
endif()

add_library(mimic_core STATIC src/ort.cpp src/faces.cpp src/emap.cpp src/mimic.cpp src/video.cpp src/picture.cpp src/stb.cpp)
target_include_directories(mimic_core PUBLIC ${STB_DIR} ${ORT_DIR} src)
if(MSVC)
    target_compile_options(mimic_core PRIVATE /O2 /fp:fast /utf-8 /D_CRT_SECURE_NO_WARNINGS)
else()
    target_link_libraries(mimic_core PUBLIC dl pthread)
endif()

add_executable(mimic-cli src/cli.cpp)
target_link_libraries(mimic-cli PRIVATE mimic_core)
if(MSVC)
    target_compile_options(mimic-cli PRIVATE /O2 /utf-8 /D_CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
    foreach(dll onnxruntime.dll DirectML.dll)
        if(EXISTS ${ORT_DIR}/${dll})
            add_custom_command(TARGET mimic-cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_DIR}/${dll} $<TARGET_FILE_DIR:mimic-cli>)
        endif()
    endforeach()
endif()

if(EXISTS ${CMAKE_SOURCE_DIR}/src/gui.cpp)
    include(FetchContent)
    if(EXISTS ${CMAKE_SOURCE_DIR}/external/imgui-1.90.4/imgui.h)
        set(FETCHCONTENT_SOURCE_DIR_IMGUI ${CMAKE_SOURCE_DIR}/external/imgui-1.90.4)
    endif()
    FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.90.4.tar.gz)
    FetchContent_MakeAvailable(imgui)
    if(WIN32 AND NOT SDL2_DIR AND EXISTS ${CMAKE_SOURCE_DIR}/external/SDL2-2.30.9/cmake)
        set(SDL2_DIR ${CMAKE_SOURCE_DIR}/external/SDL2-2.30.9/cmake)
    endif()
    find_package(SDL2 REQUIRED)
    find_package(OpenGL REQUIRED)
    add_library(imgui STATIC
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
        ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
    target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    target_link_libraries(imgui PUBLIC SDL2::SDL2 OpenGL::GL)
    set(GUI_SOURCES src/gui.cpp src/splash.cpp)
    if(WIN32 AND EXISTS ${CMAKE_SOURCE_DIR}/src/app.rc)
        list(APPEND GUI_SOURCES src/app.rc)
    endif()
    add_executable(Mimic WIN32 ${GUI_SOURCES})
    target_link_libraries(Mimic PRIVATE mimic_core imgui)
    target_compile_definitions(Mimic PRIVATE MIMIC_VERSION="${PROJECT_VERSION}")
    if(TARGET SDL2::SDL2main)
        target_link_libraries(Mimic PRIVATE SDL2::SDL2main)
    endif()
    if(MSVC)
        target_compile_options(Mimic PRIVATE /W3 /utf-8 /O2 /D_CRT_SECURE_NO_WARNINGS)
    endif()
    add_custom_command(TARGET Mimic POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:Mimic>/assets)
    if(WIN32)
        add_custom_command(TARGET Mimic POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:Mimic>)
    endif()
endif()
